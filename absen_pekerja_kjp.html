// ========================================== // KODE GOOGLE APPS SCRIPT - ABSENSI KJP // ========================================== function doGet(e) { var action = e.parameter.action; var callback = e.parameter.callback; var result = { ok: false, error: "Aksi tidak dikenal" }; try { // 1. Ambil Data Master (Proyek, Mandor, Pekerja) if (action === "get_master") { result = { ok: true, data: { proyek: getSheetDataAsJson("Proyek"), mandor: getSheetDataAsJson("Mandor"), pekerja: getSheetDataAsJson("Pekerja") } }; } // 2. Ambil Data Riwayat Absensi Berdasarkan Range Tanggal else if (action === "get_absensi") { var dari = e.parameter.dari; var sampai = e.parameter.sampai; result = { ok: true, data: getAbsensiData(dari, sampai) }; } } catch (err) { result = { ok: false, error: err.toString() }; } // Output dibungkus JSONP agar tidak terkena CORS / Blokir browser var jsonString = JSON.stringify(result); return ContentService.createTextOutput(callback + "(" + jsonString + ")") .setMimeType(ContentService.MimeType.JAVASCRIPT); } function doPost(e) { var result = { ok: false }; try { var postData = JSON.parse(e.postData.contents); var action = postData.action; var rowData = postData.row; var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Absensi"); if (!sheet) { return ContentService.createTextOutput(JSON.stringify({ ok: false, error: "Sheet 'Absensi' tidak ditemukan" })) .setMimeType(ContentService.MimeType.JSON); } // 3. Simpan Check In Baru if (action === "post_absensi") { // Cek duplikasi (Pekerja + Tanggal + Proyek) var danData = sheet.getDataRange().getValues(); var isDuplicate = false; for (var i = 1; i < danData.length; i++) { if (String(danData[i][4]) === rowData.tanggal && String(danData[i][3]) === String(rowData.pekerjaId) && String(danData[i][1]) === String(rowData.proyekId)) { isDuplicate = true; break; } } if (isDuplicate) { result = { ok: true, result: "duplicate" }; } else { sheet.appendRow([ rowData.id, rowData.proyekId, rowData.mandorId, rowData.pekerjaId, rowData.tanggal, rowData.masuk, rowData.pulang, rowData.hariKerja, rowData.lemburJam, rowData.upahHarian, rowData.upahLembur, rowData.total, rowData.deskripsi, rowData.laporanItems, rowData.foto, rowData.status, rowData.catatanReject, rowData.approvedBy, rowData.approvedAt, rowData.createdAt ]); result = { ok: true, result: "success" }; } } // 4. Update Check Out / Laporan Foto Manual else if (action === "update_absensi") { var data = sheet.getDataRange().getValues(); var targetIndex = -1; for (var i = 1; i < data.length; i++) { if (String(data[i][0]) === String(rowData.id)) { // Cocokkan ID unik absensi targetIndex = i + 1; break; } } if (targetIndex !== -1) { // Update kolom-kolom yang berubah sheet.getRange(targetIndex, 6).setValue(rowData.masuk); sheet.getRange(targetIndex, 7).setValue(rowData.pulang); sheet.getRange(targetIndex, 8).setValue(rowData.hariKerja); sheet.getRange(targetIndex, 9).setValue(rowData.lemburJam); sheet.getRange(targetIndex, 12).setValue(rowData.total); sheet.getRange(targetIndex, 13).setValue(rowData.deskripsi); sheet.getRange(targetIndex, 14).setValue(rowData.laporanItems); sheet.getRange(targetIndex, 15).setValue(rowData.foto); result = { ok: true }; } else { result = { ok: false, error: "ID Absensi tidak ditemukan di server untuk diupdate" }; } } } catch (err) { result = { ok: false, error: err.toString() }; } return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } // --- FUNGSI PEMBANTU (HELPERS) --- function getSheetDataAsJson(sheetName) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(sheetName); if (!sheet) return []; var values = sheet.getDataRange().getValues(); if (values.length <= 1) return []; var headers = values[0]; var data = []; for (var i = 1; i < values.length; i++) { var row = values[i]; var obj = {}; for (var j = 0; j < headers.length; j++) { var cellValue = row[j]; // Format Tanggal agar aman dibaca HTML if (cellValue instanceof Date) { cellValue = Utilities.formatDate(cellValue, Session.getScriptTimeZone(), "yyyy-MM-dd"); } obj[headers[j]] = cellValue; } data.push(obj); } return data; } function getAbsensiData(dari, sampai) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Absensi"); if (!sheet) return []; var values = sheet.getDataRange().getValues(); if (values.length <= 1) return []; var headers = values[0]; var data = []; for (var i = 1; i < values.length; i++) { var row = values[i]; var obj = {}; for (var j = 0; j < headers.length; j++) { var val = row[j]; if (val instanceof Date) { val = Utilities.formatDate(val, Session.getScriptTimeZone(), "yyyy-MM-dd"); } obj[headers[j]] = val; } // Filter tanggal riwayat sesuai yang diminta HTML if (obj.tanggal >= dari && obj.tanggal <= sampai) { data.push(obj); } } return data; }
