🟢 Integrasi Google Sheets

Aplikasi ini menyimpan data langsung ke Google Sheets Anda melalui Google Apps Script.
Data tidak memerlukan server tambahan — cukup satu file spreadsheet.

⚙️ Konfigurasi

Dapatkan URL ini setelah deploy Apps Script (lihat panduan di bawah)

📖 Panduan Setup Google Sheets
  1. Buka Google Sheets → buat spreadsheet baru → beri nama "Absensi KJP"
  2. Klik menu Extensions → Apps Script
  3. Hapus kode default, paste kode Apps Script di bawah ini
  4. Klik Deploy → New Deployment → Web App
  5. Execute as: Me · Who has access: Anyone
  6. Copy URL yang muncul → paste di kolom di atas → Simpan
// ═══ PT. Kaliabang Jaya Pratama — Apps Script ═══
const SS = SpreadsheetApp.getActiveSpreadsheet();

// ── GET: dipakai form pekerja untuk ambil data master ──
function doGet(e) {
  const action = e && e.parameter && e.parameter.action;
  if (action === 'get_master') {
    try {
      const proyek  = sheetToArray('Proyek',  ['id','nama','lokasi','mandorId','smId','status']);
      const mandor  = sheetToArray('Mandor',  ['id','nama','spesialis','hp']);
      const pekerja = sheetToArray('Pekerja', ['id','nama','jabatan','upah','lembur']);
      return out({ok:true, data:{proyek, mandor, pekerja}});
    } catch(err) {
      return out({ok:false, error:err.toString()});
    }
  }
  return out({ok:true, result:'pong'});
}

// ── POST: dipakai admin & form pekerja untuk sync data ──
function doPost(e) {
  try {
    const data   = JSON.parse(e.postData.contents);
    const action = data.action;
    let result;
    if (action === 'post_absensi') {
      // Form pekerja: tambah satu record absensi baru
      result = appendRow('Absensi', data.row,
        ['id','proyekId','mandorId','pekerjaId','tanggal','masuk','pulang',
         'hariKerja','lemburJam','upahHarian','upahLembur','total',
         'deskripsi','status','catatanReject','catatanEditJam','createdAt']);
    }
    else if (action === 'sync_absensi')
      result = syncSheet('Absensi', data.rows,
        ['id','proyekId','mandorId','pekerjaId','tanggal','masuk','pulang',
         'hariKerja','lemburJam','upahHarian','upahLembur','total',
         'deskripsi','status','catatanReject','catatanEditJam','approvedBy','approvedAt','createdAt']);
    else if (action === 'sync_pekerja')
      result = syncSheet('Pekerja', data.rows, ['id','nama','jabatan','upah','lembur','hp','rekening']);
    else if (action === 'sync_mandor')
      result = syncSheet('Mandor',  data.rows, ['id','nama','spesialis','hp','rekening']);
    else if (action === 'sync_proyek')
      result = syncSheet('Proyek',  data.rows, ['id','nama','lokasi','mandorId','smId','status']);
    else if (action === 'sync_all') {
      const d = data.db;
      syncSheet('Pekerja', d.pekerja, ['id','nama','jabatan','upah','lembur','hp','rekening']);
      syncSheet('Mandor',  d.mandor,  ['id','nama','spesialis','hp','rekening']);
      syncSheet('Proyek',  d.proyek,  ['id','nama','lokasi','mandorId','smId','status']);
      syncSheet('Absensi', d.absensi,
        ['id','proyekId','mandorId','pekerjaId','tanggal','masuk','pulang',
         'hariKerja','lemburJam','upahHarian','upahLembur','total',
         'deskripsi','status','catatanReject','catatanEditJam','approvedBy','approvedAt','createdAt']);
      result = 'all_synced';
    }
    else if (action === 'ping') result = 'pong';
    else result = 'unknown_action';
    return out({ok:true, result});
  } catch(err) {
    return out({ok:false, error:err.toString()});
  }
}

function out(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

// Sync penuh (timpa seluruh sheet)
function syncSheet(sheetName, rows, headers) {
  let sheet = SS.getSheetByName(sheetName);
  if (!sheet) sheet = SS.insertSheet(sheetName);
  sheet.clearContents();
  sheet.getRange(1,1,1,headers.length).setValues([headers]);
  sheet.getRange(1,1,1,headers.length)
    .setFontWeight('bold').setBackground('#C8102E').setFontColor('#ffffff');
  if (rows && rows.length > 0) {
    const vals = rows.map(r => headers.map(h => r[h] !== undefined ? r[h] : ''));
    sheet.getRange(2,1,vals.length,headers.length).setValues(vals);
  }
  sheet.autoResizeColumns(1, headers.length);
  return sheetName + '_synced_' + (rows ? rows.length : 0);
}

// Tambah satu baris (untuk post_absensi dari form pekerja)
function appendRow(sheetName, row, headers) {
  let sheet = SS.getSheetByName(sheetName);
  if (!sheet) {
    sheet = SS.insertSheet(sheetName);
    sheet.getRange(1,1,1,headers.length).setValues([headers]);
    sheet.getRange(1,1,1,headers.length)
      .setFontWeight('bold').setBackground('#C8102E').setFontColor('#ffffff');
  }
  const vals = headers.map(h => row[h] !== undefined ? row[h] : '');
  sheet.appendRow(vals);
  sheet.autoResizeColumns(1, headers.length);
  return 'appended';
}

// Baca sheet jadi array of objects
function sheetToArray(sheetName, keys) {
  const sheet = SS.getSheetByName(sheetName);
  if (!sheet) return [];
  const data = sheet.getDataRange().getValues();
  if (data.length <= 1) return [];
  const headers = data[0];
  return data.slice(1).map(row => {
    const obj = {};
    headers.forEach((h, i) => { obj[h] = row[i]; });
    return obj;
  }).filter(r => r.id);
}
